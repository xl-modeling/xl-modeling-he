cmake_minimum_required(VERSION 3.16)
project(power_model_test LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(SYSTEMC_HOME "$ENV{SYSTEMC_HOME}" CACHE PATH "SystemC installation")
set(SYSTEMC_AMS_HOME "$ENV{SYSTEMC_AMS_HOME}" CACHE PATH "SystemC-AMS installation")
find_path(SYSTEMC_INCLUDE_DIR systemc HINTS "${SYSTEMC_HOME}/include" REQUIRED)
find_library(SYSTEMC_LIBRARY systemc HINTS "${SYSTEMC_HOME}/lib" "${SYSTEMC_HOME}/lib-linux64" REQUIRED)
find_path(SYSTEMC_AMS_INCLUDE_DIR systemc-ams HINTS "${SYSTEMC_AMS_HOME}/include" REQUIRED)
find_library(SYSTEMC_AMS_LIBRARY systemc-ams HINTS "${SYSTEMC_AMS_HOME}/lib" "${SYSTEMC_AMS_HOME}/lib-linux64" REQUIRED)
find_package(Threads REQUIRED)
file(GLOB MODEL_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/../source/*.cpp")
add_executable(model_tb model_tb.cpp ${MODEL_SOURCES})
target_include_directories(model_tb PRIVATE ../include "${SYSTEMC_INCLUDE_DIR}" "${SYSTEMC_AMS_INCLUDE_DIR}")
target_compile_options(model_tb PRIVATE -Wall -Wextra)
target_link_libraries(model_tb PRIVATE "${SYSTEMC_AMS_LIBRARY}" "${SYSTEMC_LIBRARY}" Threads::Threads)
