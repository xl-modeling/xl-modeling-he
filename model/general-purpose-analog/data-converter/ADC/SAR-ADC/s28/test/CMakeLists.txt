cmake_minimum_required(VERSION 3.16)
project(s28_systemc_model LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(SYSTEMC_HOME "$ENV{SYSTEMC_HOME}" CACHE PATH "SystemC 2.3.4 installation")
set(SYSTEMC_AMS_HOME "$ENV{SYSTEMC_AMS_HOME}" CACHE PATH "SystemC AMS 2.3.4 installation")
if(NOT SYSTEMC_HOME)
    if(NOT ROOT)
        set(ROOT /mnt/d/systemc)
    endif()
    set(SYSTEMC_HOME "${ROOT}/install/systemc")
endif()
if(NOT SYSTEMC_AMS_HOME)
    if(NOT ROOT)
        set(ROOT /mnt/d/systemc)
    endif()
    set(SYSTEMC_AMS_HOME "${ROOT}/install/systemc-ams")
endif()
find_path(SC_INCLUDE systemc HINTS "${SYSTEMC_HOME}/include")
find_path(AMS_INCLUDE systemc-ams HINTS "${SYSTEMC_AMS_HOME}/include")
find_library(SC_LIBRARY NAMES systemc HINTS "${SYSTEMC_HOME}/lib" "${SYSTEMC_HOME}/lib64" "${SYSTEMC_HOME}/lib-linux64")
find_library(AMS_LIBRARY NAMES systemc-ams HINTS "${SYSTEMC_AMS_HOME}/lib" "${SYSTEMC_AMS_HOME}/lib64" "${SYSTEMC_AMS_HOME}/lib-linux64")
if(NOT SC_INCLUDE OR NOT AMS_INCLUDE OR NOT SC_LIBRARY OR NOT AMS_LIBRARY)
    message(FATAL_ERROR "Set SYSTEMC_HOME and SYSTEMC_AMS_HOME to the C++17 installations")
endif()
find_package(Threads REQUIRED)
# A static SystemC(-AMS) installation may contain non-PIC objects. Keep
# those objects in the final executable rather than embedding them in a DSO.
if(AMS_LIBRARY MATCHES "\\.a$" OR SC_LIBRARY MATCHES "\\.a$")
    set(S28_LIBRARY_TYPE STATIC)
else()
    set(S28_LIBRARY_TYPE SHARED)
endif()
message(STATUS "S28 model library type: ${S28_LIBRARY_TYPE}")
add_library(s28 ${S28_LIBRARY_TYPE} ../source/s28.cpp)
target_include_directories(s28 PUBLIC ../include "${SC_INCLUDE}" "${AMS_INCLUDE}")
target_link_libraries(s28 PUBLIC "${AMS_LIBRARY}" "${SC_LIBRARY}" Threads::Threads)
target_compile_options(s28 PRIVATE -Wall -Wextra -Wpedantic)
add_executable(s28_tb s28_tb.cpp)
target_link_libraries(s28_tb PRIVATE s28)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND S28_LIBRARY_TYPE STREQUAL "STATIC")
    # Some prebuilt non-PIC archives also contain absolute relocations that
    # cannot be used in a PIE executable. A regular ELF executable supports them.
    target_link_options(s28_tb PRIVATE -no-pie)
endif()
target_compile_options(s28_tb PRIVATE -Wall -Wextra -Wpedantic)
set_target_properties(s28 s28_tb PROPERTIES BUILD_RPATH "${SYSTEMC_HOME}/lib;${SYSTEMC_HOME}/lib64;${SYSTEMC_HOME}/lib-linux64;${SYSTEMC_AMS_HOME}/lib;${SYSTEMC_AMS_HOME}/lib64;${SYSTEMC_AMS_HOME}/lib-linux64")
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/s28_linkage.json" CONTENT
    "{\"model_library_type\":\"${S28_LIBRARY_TYPE}\",\"model_library\":\"$<TARGET_FILE_NAME:s28>\"}\n")
